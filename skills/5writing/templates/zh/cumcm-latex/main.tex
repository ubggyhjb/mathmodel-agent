%  字体依赖（正文宋体 / 标题黑体 / 强调楷体）：
%    - 按本机字体自动选择：Windows 用 SimSun（宋体）/SimHei（黑体）/KaiTi（楷体）；
%      macOS 自动回退 Songti SC / Heiti SC / Kaiti SC（fontset=none + 下方显式定义）。
%    - 西文正文：Times New Roman；代码等宽：Menlo（缺失时回退到默认字体）。
%  编译方式：xelatex main.tex （需运行两遍以生成目录）

\documentclass[fontset=none, 12pt, a4paper]{ctexart}

% ---------- 页面：A4，边距按官方优秀论文实证（左 2.9cm/上 2.8cm 中位，2026 规范要求 ≥2.5cm） ----------
\usepackage[a4paper, top=2.8cm, bottom=2.8cm, left=3cm, right=3cm]{geometry}

% ---------- 断行救急：防止行内不可断 token（代码路径/编号/±20% 串）硬越界 ----------
% ctex 中文两端对齐的字符间胶水是刚性的，行尾挤不下不可断 token 时 LaTeX 会硬凸出行宽；
% \emergencystretch 给断行器 2em 额外弹性，在判定 overfull 前先收紧相邻行。零副作用。
% 对超出 2em 的巨型 token（如 P0393/P0526 这类长编号串、长代码路径），写作时手动插断点：
%   $\{P0393,\allowbreak P0526\}$、\texttt{code/}\allowbreak 目录
\emergencystretch=2em

% ---------- 数学 ----------
\usepackage{amsmath}

% ---------- 字体（fontspec + ctex，xelatex 编译） ----------
% 中文宋体/黑体/楷体按本机字体自动选择：Windows → SimSun/SimHei/KaiTi，
% macOS 回退 Songti SC/Heiti SC/Kaiti SC（fontset=none，由下方显式定义 \songti/\heiti/\kaishu）。
\IfFontExistsTF{SimSun}{\setCJKmainfont{SimSun}}{\setCJKmainfont{Songti SC}}
\IfFontExistsTF{SimSun}{\newCJKfontfamily\songti{SimSun}}{\newCJKfontfamily\songti{Songti SC}}
\IfFontExistsTF{SimHei}{\newCJKfontfamily\heiti{SimHei}}{\newCJKfontfamily\heiti{Heiti SC}}
\IfFontExistsTF{KaiTi}{\newCJKfontfamily\kaishu{KaiTi}}{\newCJKfontfamily\kaishu{Kaiti SC}}
% 西文正文与代码等宽字体：
\IfFontExistsTF{Times New Roman}{\setmainfont{Times New Roman}}{}
\IfFontExistsTF{Menlo}{\setmonofont{Menlo}}{%
  \IfFontExistsTF{Consolas}{\setmonofont{Consolas}}{}} % Windows 回退 Consolas：覆盖希腊/≈，避免附录源码 Missing character

% ---------- 正文：12.05pt，首行缩进 2em，两端对齐 ----------
% 行距实证：官方优秀论文行距/字号比 1.30-1.42（2023 年中位 1.42）；
% \linespread 1.25 => 行距比 ≈ 1.50，介于官方与舒适度之间
\linespread{1.25}
\setlength{\parindent}{2em}

% ---------- 标题编号（\ctexset 配置） ----------
% 一级标题用中文数字“一、二、三、”，二级及以下用 1.1 / 1.1.1
\ctexset{
  section/number       = \chinese{section},
  subsection/number    = \arabic{section}.\arabic{subsection},
  subsubsection/number = \arabic{section}.\arabic{subsection}.\arabic{subsubsection},
}

% ---------- 标题格式（titlesec 配置） ----------
\usepackage{titlesec}
\titleformat{\section}
  {\centering\fontsize{17.3pt}{20.8pt}\heiti\bfseries}
  {\chinese{section}、}{1em}{}
\titleformat{\subsection}
  {\fontsize{14.45pt}{17.4pt}\heiti\bfseries}
  {\arabic{section}.\arabic{subsection}}{1em}{}
\titleformat{\subsubsection}
  {\fontsize{12.05pt}{14.5pt}\heiti\bfseries}
  {\arabic{section}.\arabic{subsection}.\arabic{subsubsection}}{1em}{}
\titlespacing{\section}       {0pt}{1.25em}{0.82em}
\titlespacing{\subsection}    {0pt}{1.15em}{0.55em}
\titlespacing{\subsubsection} {0pt}{1.15em}{0.55em}

% ---------- 自定义命令（对应 Typst 函数） ----------
% \papertitle{标题}：居中 17.3pt 加粗标题
\newcommand{\papertitle}[1]{%
  \begin{center}
    {\fontsize{17.3pt}{22pt}\bfseries #1}
  \end{center}
  \vspace{1em}
}
% \abstractcn{摘要正文}{关键词}：居中“摘要”标题（14pt 黑体）+ 正文 + “关键词：”黑体 + 关键词 + 换页
% 2026 规范：摘要页（标题+摘要+关键词）为第 1 页，页码“1”页脚居中
% 注意：标题必须用 \centering + \par，不能用 center 环境——center 的 trivlist 钩子会吞掉正文首段缩进。
\newcommand{\abstractcn}[2]{%
  \setcounter{page}{1}%
  \vspace{0.5em}%
  {\centering\fontsize{14pt}{17pt}\heiti\bfseries 摘要\par}%
  \vspace{0.5em}%
  {\sloppy #1\par}%
  \vspace{1em}
  {\heiti\bfseries 关键词：}#2\par
  \newpage
}
% \kw{...}：内容性加粗（中文黑体 + 西文加粗）。官方展示论文通行做法，三类对象才允许使用：
%   ① 最终答案（“建议时点为 16.3/15.0/15.2/22.0 周”）；② 最关键模型结论（“高 BMI 孕妇浓度整体偏低、达标更晚”）；
%   ③ 需要评委记住的性能指标（“AUPRC 提升 4.1 倍”）。
%   禁止：普通 p 值/统计量加粗、方法名单独加粗、一串裸数字逐个加粗；一句话最多 1-2 处 \kw。
% 关键数值必须包进结论短语整体加粗，每问 ≤2 处；摘要加粗率 5-15%；正文关键结论句密度 1-4%。
% 自查：删掉某个加粗后句意不受损，它就不该加粗。只用于内容强调，标题/标签不要用 \kw。
\newcommand{\kw}[1]{{\heiti\bfseries #1}}
